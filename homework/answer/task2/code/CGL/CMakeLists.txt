cmake_minimum_required(VERSION 3.13...3.31)
project(CGL LANGUAGES C CXX)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static dependencies" FORCE)
set(FT_DISABLE_ZLIB ON CACHE BOOL "Internal zlib" FORCE)
set(FT_DISABLE_BZIP2 ON CACHE BOOL "Embedded font" FORCE)
set(FT_DISABLE_PNG ON CACHE BOOL "Embedded font" FORCE)
set(FT_DISABLE_HARFBUZZ ON CACHE BOOL "Embedded font" FORCE)
set(FT_DISABLE_BROTLI ON CACHE BOOL "Embedded font" FORCE)
set(SKIP_INSTALL_ALL ON CACHE BOOL "Embedded dependency" FORCE)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "Build GLFW library" FORCE)
add_subdirectory(deps/freetype/freetype)
add_subdirectory(deps/glew)
add_subdirectory(deps/glfw)
set(CGL_SOURCE src/vector2D.cpp src/vector3D.cpp src/vector4D.cpp src/matrix3x3.cpp
 src/matrix4x4.cpp src/quaternion.cpp src/complex.cpp src/color.cpp src/osdtext.cpp
 src/osdfont.cpp src/viewer.cpp src/base64.cpp src/tinyxml2.cpp)
add_library(CGL STATIC ${CGL_SOURCE})
target_include_directories(CGL PUBLIC include include/CGL deps/glew/include deps/glfw/include)
target_compile_definitions(CGL PUBLIC GLEW_STATIC)
target_link_libraries(CGL PUBLIC glew glfw OpenGL::GL PRIVATE freetype)
