cmake_minimum_required(VERSION 3.13...3.31)
project(Assignment2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(OpenGL_GL_PREFERENCE LEGACY)
find_package(OpenGL REQUIRED)
add_subdirectory(CGL)
set(MESH_SOURCES src/bezierPatch.cpp src/bezierCurve.cpp src/scene.cpp src/camera.cpp
  src/light.cpp src/mesh.cpp src/material.cpp src/texture.cpp src/collada.cpp
  src/halfEdgeMesh.cpp src/student_code.cpp src/meshEdit.cpp src/png.cpp src/lodepng.cpp)
add_library(mesh_core STATIC ${MESH_SOURCES})
target_include_directories(mesh_core PUBLIC src)
target_link_libraries(mesh_core PUBLIC CGL OpenGL::GL OpenGL::GLU)
target_compile_definitions(mesh_core PUBLIC GLEW_STATIC)
add_executable(meshedit src/main.cpp)
target_link_libraries(meshedit PRIVATE mesh_core)
option(BUILD_VERIFICATION "Build algorithm verification" OFF)
if(BUILD_VERIFICATION)
  enable_testing()
  add_executable(geometry_tests verification/geometry_tests.cpp)
  target_link_libraries(geometry_tests PRIVATE mesh_core)
  add_test(NAME geometry_algorithms COMMAND geometry_tests)
  foreach(scene cube teapot open-square)
    add_test(NAME geometry_${scene} COMMAND geometry_tests ${CMAKE_CURRENT_SOURCE_DIR}/dae/${scene}.dae)
  endforeach()
  add_test(NAME geometry_humanoid COMMAND geometry_tests ${CMAKE_CURRENT_SOURCE_DIR}/dae/custom/humanoid.dae)
endif()
